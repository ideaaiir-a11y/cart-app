    73→function csvCell(v: string | number): string {
    74→  const s = String(v ?? "");
    75→  return `"${s.replace(/"/g, '""')}"`;
    76→}
    77→
    78→function buildCardsCsv(cards: CardWithHtml[]): string {
    79→  const header = [
    80→    "نام محصول",
    81→    "نام لاتین",
    82→    "قیمت (تومان)",
    83→    "لینک مرجع",
    84→    "تم قالب",
    85→    "تعداد مشخصات",
    86→    "روش مصرف",
    87→  ];
    88→  const rows = cards.map((c) => [
    89→    c.name || c.rawInput,
    90→    c.nameEn || "",
    91→    c.priceValue || 0,
    92→    c.link || "",
    93→    themeLabel(c.theme),
    94→    c.specs?.length ?? 0,
    95→    (c.usage || "").replace(/\r?\n/g, " | "),
    96→  ]);
    97→  const body = [header, ...rows].map((r) => r.map(csvCell).join(",")).join("\r\n");
    98→  return `\uFEFF${body}`;
    99→}

   138→function VerifyBadges({ card }: { card: CardWithHtml }) {
   139→  if (!card.issues || card.issues.length === 0) return null;
   140→  return (
   141→    <div className="space-y-2">
   142→      <h4 className="text-sm font-bold">راستی‌آزمایی</h4>
   143→      <div className="flex flex-wrap gap-2">
   144→        {card.issues.map((issue, i) => {
   145→          const tone = issue.ok
   146→            ? "border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-500/30 dark:bg-emerald-950/40 dark:text-emerald-300"
   147→            : issue.fixed

   149→              : "border-red-300 bg-red-50 text-red-600 dark:border-red-500/30 dark:bg-red-950/40 dark:text-red-300";

   151→            <Tooltip key={i}>
   152→              <TooltipTrigger asChild>
   153→                <Badge variant="outline" className={`cursor-default gap-1.5 rounded-full px-3 py-1.5 ${tone}`}>
   154→                  {issue.item}
   155→                  {issue.ok ? (
   156→                    <Check className="size-3.5" aria-hidden />
   157→                  ) : issue.fixed ? (
   158→                    <Check className="size-3.5" aria-hidden />
   159→                  ) : (
   160→                    <X className="size-3.5" aria-hidden />
   161→                  )}

   165→              <TooltipContent className="max-w-64 text-xs" dir="rtl">
   166→                {issue.detail}
   167→              </TooltipContent>

   427→/* ─── اعمال گروهی تم روی همه کارت‌ها ──────────────────── */
   428→
   429→function BulkThemePopover({
   430→  jobId,
   431→  disabled,
   432→  onApplied,
   433→}: {
   434→  jobId: string;
   435→  disabled: boolean;
   436→  onApplied: () => void;
   437→}) {
   438→  const [open, setOpen] = React.useState(false);
   439→  const [applying, setApplying] = React.useState("");
   440→
   441→  const applyTheme = async (key: string, label: string) => {
   442→    setApplying(key);
   443→    try {
   444→      const data = await apiJson<{ ok: boolean; updated: number; theme: string }>(
   445→        `/api/jobs/${jobId}/theme`,
   446→        {
   447→          method: "POST",
   448→          headers: { "Content-Type": "application/json" },
   449→          body: JSON.stringify({ theme: key }),
   450→        },
   451→      );
   452→      toast.success(`تم «${label}» روی ${faNum(data.updated ?? 0)} کارت اعمال شد`);
   453→      setOpen(false);
   454→      onApplied();
   455→    } catch (e) {
   456→      toast.error("اعمال گروهی تم ناموفق بود", { description: toErrorMessage(e) });
   457→    } finally {
   458→      setApplying("");
   459→    }
   460→  };
   461→
   462→  return (
   463→    <Popover open={open} onOpenChange={setOpen}>
   464→      <PopoverTrigger asChild>
   465→        <Button
   466→          variant="outline"
   467→          disabled={disabled}
   468→          className="h-11 gap-2 rounded-xl border-emerald-300 bg-white/70 text-emerald-800 hover:border-emerald-500 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-500/30 dark:bg-emerald-950/20 dark:text-emerald-300"
   469→        >
   470→          <PaintBucket className="size-5" aria-hidden />
   471→          تغییر گروهی تم
   472→        </Button>
   473→      </PopoverTrigger>
   474→      <PopoverContent dir="rtl" align="center" className="w-72 rounded-xl p-3">
   475→        <p className="mb-2 text-xs font-bold">انتخاب تم برای همه کارت‌های این پروسه</p>
   476→        <p className="mb-3 text-[11px] leading-5 text-muted-foreground">
   477→          HTML همه کارت‌ها بلافاصله با تم جدید بازسازی می‌شود.
   478→        </p>
   479→        <div className="grid grid-cols-2 gap-2">
   480→          {CARD_THEME_OPTIONS.map((t) => (
   481→            <button
   482→              key={t.key}
   483→              type="button"
   484→              disabled={applying !== ""}
   485→              onClick={() => void applyTheme(t.key, t.label)}
   486→              className="flex items-center gap-2 rounded-lg border px-2.5 py-2 text-right text-xs font-semibold transition-all hover:-translate-y-0.5 hover:border-emerald-400 hover:bg-emerald-50/60 disabled:opacity-50 dark:hover:bg-emerald-950/30"
   487→            >
   488→              {applying === t.key ? (
   489→                <Loader2 className="size-4 shrink-0 animate-spin text-emerald-600" aria-hidden />
   490→              ) : (
   491→                <span
   492→                  className="size-4 shrink-0 rounded-full ring-1 ring-black/10"
   493→                  style={{ background: t.swatch[0] }}
   494→                  aria-hidden
   495→                />
   496→              )}
   497→              {t.label}
   498→            </button>
   499→          ))}
   500→        </div>
   501→      </PopoverContent>
   502→    </Popover>
   503→  );
   504→}

   508→export interface OutputGalleryProps {
   509→  jobId: string;
   510→  cards: CardWithHtml[];
   511→  running: boolean;
   512→  onEditCard: (card: CardWithHtml) => void;
   513→  onCardUpdated: (card: CardWithHtml) => void;
   514→  onBulkThemeApplied: () => void;
   515→}
   516→
   517→export function OutputGallery({
   518→  jobId,
   519→  cards,
   520→  running,
   521→  onEditCard,
   522→  onCardUpdated,
   523→  onBulkThemeApplied,
   524→}: OutputGalleryProps) {
   525→  const [query, setQuery] = React.useState("");
   526→  const [sortBy, setSortBy] = React.useState("default");
   527→
   528→  const visibleCards = React.useMemo(() => {
   529→    const q = query.trim().toLowerCase();
   530→    let list = cards;
   531→    if (q) {
   532→      list = list.filter((c) =>
   533→        `${c.name || ""} ${c.nameEn || ""} ${c.rawInput}`
   534→          .toLowerCase()
   535→          .includes(q),
   536→      );
   537→    }
   538→    if (sortBy === "name") {
   539→      list = [...list].sort((a, b) =>
   540→        (a.name || a.rawInput).localeCompare(b.name || b.rawInput, "fa"),
   541→      );
   542→    } else if (sortBy === "price-asc") {
   543→      list = [...list].sort((a, b) => (a.priceValue || 0) - (b.priceValue || 0));
   544→    } else if (sortBy === "price-desc") {
   545→      list = [...list].sort((a, b) => (b.priceValue || 0) - (a.priceValue || 0));
   546→    }
   547→    return list;
   548→  }, [cards, query, sortBy]);
   549→
   550→  const exportCsv = () => {
   551→    if (!visibleCards.length) {
   552→      toast.error("کارتی برای خروجی CSV وجود ندارد");
   553→      return;
   554→    }
   555→    try {
   556→      const csv = buildCardsCsv(visibleCards);
   557→      const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   558→      const url = URL.createObjectURL(blob);
   559→      const a = document.createElement("a");
   560→      a.href = url;
   561→      a.download = `cards-${jobId.slice(-6)}.csv`;
   562→      document.body.appendChild(a);
   563→      a.click();
   564→      a.remove();
   565→      URL.revokeObjectURL(url);
   566→      toast.success(`فایل CSV با ${faNum(visibleCards.length)} کارت دانلود شد`);
   567→    } catch (e) {
   568→      toast.error("ساخت فایل CSV ناموفق بود", { description: toErrorMessage(e) });
   569→    }
   570→  };
   571→
   572→  if (!running && cards.length === 0) return null;
   573→
   574→  return (
   575→    <TooltipProvider delayDuration={200}>
   576→      <motion.section
   577→        id="outputs"
   578→        initial={{ opacity: 0, y: 24 }}
   579→        animate={{ opacity: 1, y: 0 }}
   580→        transition={{ duration: 0.4, ease: "easeOut" }}
   581→        className="scroll-mt-24 space-y-5"
   582→      >
   583→        <div className="flex flex-col items-center gap-3 rounded-2xl border border-dashed border-emerald-300 bg-emerald-50/50 p-5 text-center dark:border-emerald-500/30 dark:bg-emerald-950/20">
   584→          <h2 className="text-lg font-extrabold">
   585→            خروجی‌ها
   586→            <Badge className="mr-2 rounded-full bg-emerald-600 px-2.5 py-0.5 text-white">
   587→              {faNum(cards.length)} کارت
   588→            </Badge>
   589→          </h2>
   590→          <p className="text-xs text-muted-foreground">
   591→            پیش‌نمایش زنده کارت‌های تولیدشده — دانلود HTML، JSON ساختاریافته یا ZIP کامل (شامل HTML + JSON + cards.json)
   592→          </p>
   593→          {cards.length > 0 && (
   594→            <div className="flex flex-wrap items-center justify-center gap-2.5">
   595→              <Button
   596→                asChild
   597→                size="lg"
   598→                className="h-11 gap-2 rounded-xl bg-emerald-600 text-white shadow-md shadow-emerald-600/25 hover:bg-emerald-700"
   599→              >
   600→                <a href={`/api/jobs/${jobId}/download`} download>
   601→                  <FileArchive className="size-5" aria-hidden />
   602→                  دانلود ZIP همه کارت‌ها
   603→                </a>
   604→              </Button>
   605→              <BulkThemePopover jobId={jobId} disabled={running} onApplied={onBulkThemeApplied} />
   606→              <Button
   607→                variant="outline"
   608→                onClick={() => void exportCsv()}
   609→                disabled={running}
   610→                title="جدول قیمت و مشخصات همه کارت‌ها برای اکسل"
   611→                className="h-11 gap-2 rounded-xl border-emerald-300 bg-white/70 text-emerald-800 hover:border-emerald-500 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-500/30 dark:bg-emerald-950/20 dark:text-emerald-300"
   612→              >
   613→                <FileSpreadsheet className="size-5" aria-hidden />
   614→                خروجی CSV
   615→              </Button>
   616→            </div>
   617→          )}
   618→          {cards.length > 2 && (
   619→            <div className="flex w-full max-w-xl flex-col gap-2 sm:flex-row sm:items-center">
   620→              <div className="relative flex-1">
   621→                <Search
   622→                  className="pointer-events-none absolute inset-y-0 right-3 flex w-4 items-center text-muted-foreground"
   623→                  aria-hidden
   624→                />
   625→                <Input
   626→                  value={query}
   627→                  onChange={(e) => setQuery(e.target.value)}
   628→                  placeholder="جستجو در کارت‌ها (نام فارسی یا لاتین)…"
   629→                  className="h-10 rounded-xl pr-9"
   630→                  aria-label="جستجو در کارت‌های خروجی"
   631→                />
   632→                {query && (
   633→                  <button
   634→                    type="button"
   635→                    onClick={() => setQuery("")}
   636→                    aria-label="پاک کردن جستجو"
   637→                    className="absolute inset-y-0 left-2 flex w-7 items-center justify-center rounded-md text-muted-foreground transition-colors hover:text-foreground"
   638→                  >
   639→                    <X className="size-4" aria-hidden />
   640→                  </button>
   641→                )}
   642→              </div>
   643→              <Select dir="rtl" value={sortBy} onValueChange={setSortBy}>
   644→                <SelectTrigger className="h-10 w-full rounded-xl sm:w-52" aria-label="مرتب‌سازی کارت‌ها">
   645→                  <span className="flex items-center gap-1.5 text-xs">
   646→                    <ArrowDownWideNarrow className="size-4 text-emerald-600 dark:text-emerald-400" aria-hidden />
   647→                    <SelectValue />
   648→                  </span>
   649→                </SelectTrigger>
   650→                <SelectContent>
   651→                  {SORT_OPTIONS.map((o) => (
   652→                    <SelectItem key={o.value} value={o.value}>
   653→                      {o.label}
   654→                    </SelectItem>
   655→                  ))}
   656→                </SelectContent>
   657→              </Select>
   658→            </div>
   659→          )}
   660→        </div>
   661→
   662→        {cards.length === 0 && running ? (
   663→          <div className="grid grid-cols-1 gap-6 md:grid-cols-2 xl:grid-cols-3">
   664→            {[0, 1, 2].map((i) => (
   665→              <div key={i} className="overflow-hidden rounded-2xl border bg-card">
   666→                <Skeleton className="h-[420px] w-full rounded-none" />
   667→                <div className="space-y-3 p-4">
   668→                  <Skeleton className="h-5 w-3/4" />
   669→                  <Skeleton className="h-6 w-24 rounded-full" />
   670→                  <div className="grid grid-cols-2 gap-2">
   671→                    <Skeleton className="h-10 rounded-xl" />
   672→                    <Skeleton className="h-10 rounded-xl" />
   673→                    <Skeleton className="h-10 rounded-xl" />
   674→                    <Skeleton className="h-10 rounded-xl" />
   675→                  </div>
   676→                </div>
   677→              </div>
   678→            ))}
   679→          </div>
   680→        ) : visibleCards.length === 0 ? (
   681→          <div className="flex flex-col items-center gap-2.5 rounded-2xl border border-dashed p-10 text-center">
   682→            <Search className="size-10 text-emerald-400/70" aria-hidden />
   683→            <p className="text-sm font-medium">کارتی با عبارت «{query}» یافت نشد</p>
   684→            <Button
   685→              variant="ghost"
   686→              size="sm"
   687→              onClick={() => {
   688→                setQuery("");
   689→                setSortBy("default");
   690→              }}
   691→              className="h-9 rounded-lg text-xs text-emerald-700 hover:text-emerald-800 dark:text-emerald-400"
   692→            >
   693→              پاک کردن جستجو و مرتب‌سازی
   694→            </Button>
   695→          </div>
   696→        ) : (
   697→          <div className="grid grid-cols-1 gap-6 md:grid-cols-2 xl:grid-cols-3">
   698→            {visibleCards.map((card, i) => (
   699→              <OutputCardItem
   700→                key={card.id}
   701→                card={card}
   702→                jobId={jobId}
   703→                index={i}
   704→                onEdit={onEditCard}
   705→                onCardUpdated={onCardUpdated}
   706→              />
   707→            ))}
   708→          </div>
   709→        )}
   710→      </motion.section>
   711→    </TooltipProvider>
   712→  );
   713→}
