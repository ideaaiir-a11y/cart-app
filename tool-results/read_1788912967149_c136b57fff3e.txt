    76→function csvCell(v: string | number): string {
    77→  const s = String(v ?? "");
    78→  return `"${s.replace(/"/g, '""')}"`;

    81→function buildCardsCsv(cards: CardWithHtml[]): string {
    82→  const header = [
    83→    "نام محصول",
    84→    "نام لاتین",
    85→    "قیمت (تومان)",
    86→    "لینک مرجع",
    87→    "تم قالب",
    88→    "تعداد مشخصات",
    89→    "روش مصرف",
    90→  ];
    91→  const rows = cards.map((c) => [
    92→    c.name || c.rawInput,
    93→    c.nameEn || "",
    94→    c.priceValue || 0,
    95→    c.link || "",
    96→    themeLabel(c.theme),
    97→    c.specs?.length ?? 0,
    98→    (c.usage || "").replace(/\r?\n/g, " | "),
    99→  ]);
   100→  const body = [header, ...rows].map((r) => r.map(csvCell).join(",")).join("\r\n");
   101→  return `\uFEFF${body}`;
   102→}

   141→function VerifyBadges({ card }: { card: CardWithHtml }) {
   142→  if (!card.issues || card.issues.length === 0) return null;
   143→  return (
   144→    <div className="space-y-2">
   145→      <h4 className="text-sm font-bold">راستی‌آزمایی</h4>
   146→      <div className="flex flex-wrap gap-2">
   147→        {card.issues.map((issue, i) => {
   148→          const tone = issue.ok
   149→            ? "border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-500/30 dark:bg-emerald-950/40 dark:text-emerald-300"
   150→            : issue.fixed
   151→              ? "border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-500/30 dark:bg-amber-950/40 dark:text-amber-300"
   152→              : "border-red-300 bg-red-50 text-red-600 dark:border-red-500/30 dark:bg-red-950/40 dark:text-red-300";
   153→          return (
   154→            <Tooltip key={i}>
   155→              <TooltipTrigger asChild>
   156→                <Badge variant="outline" className={`cursor-default gap-1.5 rounded-full px-3 py-1.5 ${tone}`}>
   157→                  {issue.item}
   158→                  {issue.ok ? (
   159→                    <Check className="size-3.5" aria-hidden />
   160→                  ) : issue.fixed ? (
   161→                    <Check className="size-3.5" aria-hidden />
   162→                  ) : (
   163→                    <X className="size-3.5" aria-hidden />
   164→                  )}
   165→                  {issue.fixed && !issue.ok && <span className="text-[10px]">(اصلاح شد)</span>}
   166→                </Badge>
   167→              </TooltipTrigger>
   168→              <TooltipContent className="max-w-64 text-xs" dir="rtl">
   169→                {issue.detail}
   170→              </TooltipContent>
   171→            </Tooltip>
   172→          );
   173→        })}
   174→      </div>
   175→    </div>
   176→  );
   177→}

   547→/* ─── اعمال گروهی تم روی همه کارت‌ها ──────────────────── */
   548→
   549→function BulkThemePopover({
   550→  jobId,
   551→  disabled,
   552→  onApplied,
   553→}: {
   554→  jobId: string;
   555→  disabled: boolean;
   556→  onApplied: () => void;
   557→}) {
   558→  const [open, setOpen] = React.useState(false);
   559→  const [applying, setApplying] = React.useState("");
   560→
   561→  const applyTheme = async (key: string, label: string) => {
   562→    setApplying(key);
   563→    try {
   564→      const data = await apiJson<{ ok: boolean; updated: number; theme: string }>(
   565→        `/api/jobs/${jobId}/theme`,
   566→        {
   567→          method: "POST",
   568→          headers: { "Content-Type": "application/json" },
   569→          body: JSON.stringify({ theme: key }),
   570→        },
   571→      );
   572→      toast.success(`تم «${label}» روی ${faNum(data.updated ?? 0)} کارت اعمال شد`);
   573→      setOpen(false);
   574→      onApplied();
   575→    } catch (e) {
   576→      toast.error("اعمال گروهی تم ناموفق بود", { description: toErrorMessage(e) });
   577→    } finally {
   578→      setApplying("");
   579→    }
   580→  };
   581→
   582→  return (
   583→    <Popover open={open} onOpenChange={setOpen}>
   584→      <PopoverTrigger asChild>
   585→        <Button
   586→          variant="outline"
   587→          disabled={disabled}
   588→          className="h-11 gap-2 rounded-xl border-emerald-300 bg-white/70 text-emerald-800 hover:border-emerald-500 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-500/30 dark:bg-emerald-950/20 dark:text-emerald-300"
   589→        >
   590→          <PaintBucket className="size-5" aria-hidden />
   591→          تغییر گروهی تم
   592→        </Button>
   593→      </PopoverTrigger>
   594→      <PopoverContent dir="rtl" align="center" className="w-72 rounded-xl p-3">
   595→        <p className="mb-2 text-xs font-bold">انتخاب تم برای همه کارت‌های این پروسه</p>
   596→        <p className="mb-3 text-[11px] leading-5 text-muted-foreground">
   597→          HTML همه کارت‌ها بلافاصله با تم جدید بازسازی می‌شود.
   598→        </p>
   599→        <div className="grid grid-cols-2 gap-2">
   600→          {CARD_THEME_OPTIONS.map((t) => (
   601→            <button
   602→              key={t.key}
   603→              type="button"
   604→              disabled={applying !== ""}
   605→              onClick={() => void applyTheme(t.key, t.label)}
   606→              className="flex items-center gap-2 rounded-lg border px-2.5 py-2 text-right text-xs font-semibold transition-all hover:-translate-y-0.5 hover:border-emerald-400 hover:bg-emerald-50/60 disabled:opacity-50 dark:hover:bg-emerald-950/30"
   607→            >
   608→              {applying === t.key ? (
   609→                <Loader2 className="size-4 shrink-0 animate-spin text-emerald-600" aria-hidden />
   610→              ) : (
   611→                <span
   612→                  className="size-4 shrink-0 rounded-full ring-1 ring-black/10"
   613→                  style={{ background: t.swatch[0] }}
   614→                  aria-hidden
   615→                />
   616→              )}
   617→              {t.label}
   618→            </button>
   619→          ))}
   620→        </div>
   621→      </PopoverContent>
   622→    </Popover>
   623→  );
   624→}

   627→
   628→export interface OutputGalleryProps {
   629→  jobId: string;
   630→  cards: CardWithHtml[];
   631→  running: boolean;
   632→  onEditCard: (card: CardWithHtml) => void;
   633→  onCardUpdated: (card: CardWithHtml) => void;
   634→  onBulkThemeApplied: () => void;
   635→}
   636→
   637→export function OutputGallery({
   638→  jobId,
   639→  cards,
   640→  running,
   641→  onEditCard,
   642→  onCardUpdated,
   643→  onBulkThemeApplied,
   644→}: OutputGalleryProps) {
   645→  const [query, setQuery] = React.useState("");
   646→  const [sortBy, setSortBy] = React.useState("default");
   647→
   648→  const visibleCards = React.useMemo(() => {
   649→    const q = query.trim().toLowerCase();
   650→    let list = cards;
   651→    if (q) {
   652→      list = list.filter((c) =>
   653→        `${c.name || ""} ${c.nameEn || ""} ${c.rawInput}`
   654→          .toLowerCase()
   655→          .includes(q),
   656→      );
   657→    }
   658→    if (sortBy === "name") {
   659→      list = [...list].sort((a, b) =>
   660→        (a.name || a.rawInput).localeCompare(b.name || b.rawInput, "fa"),
   661→      );
   662→    } else if (sortBy === "price-asc") {
   663→      list = [...list].sort((a, b) => (a.priceValue || 0) - (b.priceValue || 0));
   664→    } else if (sortBy === "price-desc") {
   665→      list = [...list].sort((a, b) => (b.priceValue || 0) - (a.priceValue || 0));
   666→    }
   667→    return list;
   668→  }, [cards, query, sortBy]);
   669→
   670→  const exportCsv = () => {
   671→    if (!visibleCards.length) {
   672→      toast.error("کارتی برای خروجی CSV وجود ندارد");
   673→      return;
   674→    }
   675→    try {
   676→      const csv = buildCardsCsv(visibleCards);
   677→      const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   678→      const url = URL.createObjectURL(blob);
   679→      const a = document.createElement("a");
   680→      a.href = url;
   681→      a.download = `cards-${jobId.slice(-6)}.csv`;
   682→      document.body.appendChild(a);
   683→      a.click();
   684→      a.remove();
   685→      URL.revokeObjectURL(url);
   686→      toast.success(`فایل CSV با ${faNum(visibleCards.length)} کارت دانلود شد`);
   687→    } catch (e) {
   688→      toast.error("ساخت فایل CSV ناموفق بود", { description: toErrorMessage(e) });
   689→    }
   690→  };
   691→
   692→  if (!running && cards.length === 0) return null;
   693→
   694→  return (
   695→    <TooltipProvider delayDuration={200}>
   696→      <motion.section
   697→        id="outputs"
   698→        initial={{ opacity: 0, y: 24 }}
   699→        animate={{ opacity: 1, y: 0 }}
   700→        transition={{ duration: 0.4, ease: "easeOut" }}
   701→        className="scroll-mt-24 space-y-5"
   702→      >
   703→        <div className="relative overflow-hidden rounded-2xl border border-dashed border-emerald-300 bg-gradient-to-bl from-emerald-50/80 via-white to-amber-50/60 p-5 text-center shadow-sm dark:border-emerald-500/30 dark:from-emerald-950/25 dark:via-card dark:to-amber-950/10">
   704→          <div
   705→            className="pointer-events-none absolute -left-10 -top-10 size-32 rounded-full bg-emerald-200/40 blur-2xl dark:bg-emerald-500/10"
   706→            aria-hidden
   707→          />
   708→          <div
   709→            className="pointer-events-none absolute -bottom-12 -right-8 size-32 rounded-full bg-amber-200/40 blur-2xl dark:bg-amber-500/10"
   710→            aria-hidden
   711→          />
   712→          <h2 className="relative text-lg font-extrabold">
   713→            خروجی‌ها
   714→            <Badge className="mr-2 rounded-full bg-emerald-600 px-2.5 py-0.5 text-white">
   715→              {faNum(cards.length)} کارت
   716→            </Badge>
   717→          </h2>
   718→          <p className="relative text-xs text-muted-foreground">
   719→            پیش‌نمایش زنده کارت‌های تولیدشده — دانلود HTML، JSON ساختاریافته یا ZIP کامل (شامل HTML + JSON + cards.json)
   720→          </p>
   721→          {cards.length > 0 && (
   722→            <div className="relative flex flex-wrap items-center justify-center gap-2.5">
   723→              <Button
   724→                asChild
   725→                size="lg"
   726→                className="h-11 gap-2 rounded-xl bg-emerald-600 text-white shadow-md shadow-emerald-600/25 hover:bg-emerald-700"
   727→              >
   728→                <a href={`/api/jobs/${jobId}/download`} download>
   729→                  <FileArchive className="size-5" aria-hidden />
   730→                  دانلود ZIP همه کارت‌ها
   731→                </a>
   732→              </Button>
   733→              <BulkThemePopover jobId={jobId} disabled={running} onApplied={onBulkThemeApplied} />
   734→              <Button
   735→                variant="outline"
   736→                onClick={() => void exportCsv()}
   737→                disabled={running}
   738→                title="جدول قیمت و مشخصات همه کارت‌ها برای اکسل"
   739→                className="h-11 gap-2 rounded-xl border-emerald-300 bg-white/70 text-emerald-800 hover:border-emerald-500 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-500/30 dark:bg-emerald-950/20 dark:text-emerald-300"

   752→                />
   753→                <Input
   754→                  value={query}

   756→                  placeholder="جستجو در کارت‌ها (نام فارسی یا لاتین)…"
   757→                  className="h-10 rounded-xl pr-9"

   812→            <Button
   813→              variant="ghost"
   814→              size="sm"

   816→                setQuery("");

   818→              }}
   819→              className="h-9 rounded-lg text-xs text-emerald-700 hover:text-emerald-800 dark:text-emerald-400"
   820→            >
   821→              پاک کردن جستجو و مرتب‌سازی
