    78→function csvCell(v: string | number): string {
    79→  const s = String(v ?? "");
    80→  return `"${s.replace(/"/g, '""')}"`;

    83→function buildCardsCsv(cards: CardWithHtml[]): string {
    84→  const header = [
    85→    "نام محصول",
    86→    "نام لاتین",
    87→    "قیمت (تومان)",
    88→    "لینک مرجع",
    89→    "تم قالب",
    90→    "تعداد مشخصات",
    91→    "روش مصرف",
    92→  ];
    93→  const rows = cards.map((c) => [
    94→    c.name || c.rawInput,
    95→    c.nameEn || "",
    96→    c.priceValue || 0,
    97→    c.link || "",
    98→    themeLabel(c.theme),
    99→    c.specs?.length ?? 0,
   100→    (c.usage || "").replace(/\r?\n/g, " | "),
   101→  ]);
   102→  const body = [header, ...rows].map((r) => r.map(csvCell).join(",")).join("\r\n");
   103→  return `\uFEFF${body}`;
   104→}

   142→
   143→function VerifyBadges({ card }: { card: CardWithHtml }) {
   144→  if (!card.issues || card.issues.length === 0) return null;
   145→  return (
   146→    <div className="space-y-2">
   147→      <h4 className="text-sm font-bold">راستی‌آزمایی</h4>
   148→      <div className="flex flex-wrap gap-2">
   149→        {card.issues.map((issue, i) => {
   150→          const tone = issue.ok
   151→            ? "border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-500/30 dark:bg-emerald-950/40 dark:text-emerald-300"
   152→            : issue.fixed
   153→              ? "border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-500/30 dark:bg-amber-950/40 dark:text-amber-300"
   154→              : "border-red-300 bg-red-50 text-red-600 dark:border-red-500/30 dark:bg-red-950/40 dark:text-red-300";
   155→          return (
   156→            <Tooltip key={i}>
   157→              <TooltipTrigger asChild>
   158→                <Badge variant="outline" className={`cursor-default gap-1.5 rounded-full px-3 py-1.5 ${tone}`}>
   159→                  {issue.item}
   160→                  {issue.ok ? (
   161→                    <Check className="size-3.5" aria-hidden />
   162→                  ) : issue.fixed ? (
   163→                    <Check className="size-3.5" aria-hidden />
   164→                  ) : (
   165→                    <X className="size-3.5" aria-hidden />
   166→                  )}
   167→                  {issue.fixed && !issue.ok && <span className="text-[10px]">(اصلاح شد)</span>}
   168→                </Badge>
   169→              </TooltipTrigger>
   170→              <TooltipContent className="max-w-64 text-xs" dir="rtl">
   171→                {issue.detail}
   172→              </TooltipContent>
   173→            </Tooltip>
   174→          );
   175→        })}
   176→      </div>
   177→    </div>
   178→  );
   179→}

   481→/* ─── دکمه نسخه انگلیسی کارت (ترجمه + دانلود چپ‌به‌راست) ─────────── */
   482→
   483→function EnVersionButton({
   484→  card,
   485→  jobId,
   486→  onCardUpdated,
   487→}: {
   488→  card: CardWithHtml;
   489→  jobId: string;
   490→  onCardUpdated: (c: CardWithHtml) => void;
   491→}) {
   492→  const [building, setBuilding] = React.useState(false);
   493→
   494→  const downloadEn = () => {
   495→    const a = document.createElement("a");
   496→    a.href = `/api/jobs/${jobId}/download/${card.id}?lang=en`;
   497→    a.download = "";
   498→    document.body.appendChild(a);
   499→    a.click();
   500→    a.remove();
   501→  };
   502→
   503→  const buildEn = async () => {
   504→    setBuilding(true);
   505→    try {
   506→      await apiJson<{ ok: boolean }>(`/api/cards/${card.id}/translate`, {
   507→        method: "POST",
   508→      });
   509→      onCardUpdated({ ...card, hasEn: true });
   510→      toast.success("نسخه انگلیسی ساخته شد؛ دانلود آغاز می‌شود");
   511→      downloadEn();
   512→    } catch (e) {
   513→      toast.error("ساخت نسخه انگلیسی ناموفق بود", {
   514→        description: toErrorMessage(e),
   515→      });
   516→    } finally {
   517→      setBuilding(false);
   518→    }

   684→
   685→function BulkThemePopover({
   686→  jobId,
   687→  disabled,
   688→  onApplied,
   689→}: {
   690→  jobId: string;
   691→  disabled: boolean;
   692→  onApplied: () => void;
   693→}) {
   694→  const [open, setOpen] = React.useState(false);
   695→  const [applying, setApplying] = React.useState("");
   696→
   697→  const applyTheme = async (key: string, label: string) => {
   698→    setApplying(key);
   699→    try {
   700→      const data = await apiJson<{ ok: boolean; updated: number; theme: string }>(
   701→        `/api/jobs/${jobId}/theme`,
   702→        {
   703→          method: "POST",
   704→          headers: { "Content-Type": "application/json" },
   705→          body: JSON.stringify({ theme: key }),
   706→        },
   707→      );
   708→      toast.success(`تم «${label}» روی ${faNum(data.updated ?? 0)} کارت اعمال شد`);
   709→      setOpen(false);
   710→      onApplied();
   711→    } catch (e) {
   712→      toast.error("اعمال گروهی تم ناموفق بود", { description: toErrorMessage(e) });
   713→    } finally {
   714→      setApplying("");

   716→  };

   718→  return (
   719→    <Popover open={open} onOpenChange={setOpen}>
   720→      <PopoverTrigger asChild>
   721→        <Button
   722→          variant="outline"
   723→          disabled={disabled}
   724→          className="h-11 gap-2 rounded-xl border-emerald-300 bg-white/70 text-emerald-800 hover:border-emerald-500 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-500/30 dark:bg-emerald-950/20 dark:text-emerald-300"
   725→        >
   726→          <PaintBucket className="size-5" aria-hidden />
   727→          تغییر گروهی تم
   728→        </Button>

   732→        <p className="mb-3 text-[11px] leading-5 text-muted-foreground">
   733→          HTML همه کارت‌ها بلافاصله با تم جدید بازسازی می‌شود.

   737→            <button
   738→              key={t.key}

   741→              onClick={() => void applyTheme(t.key, t.label)}
   742→              className="flex items-center gap-2 rounded-lg border px-2.5 py-2 text-right text-xs font-semibold transition-all hover:-translate-y-0.5 hover:border-emerald-400 hover:bg-emerald-50/60 disabled:opacity-50 dark:hover:bg-emerald-950/30"

   781→  const [query, setQuery] = React.useState("");
   782→  const [sortBy, setSortBy] = React.useState("default");
   783→
   784→  const visibleCards = React.useMemo(() => {
   785→    const q = query.trim().toLowerCase();
   786→    let list = cards;
   787→    if (q) {
   788→      list = list.filter((c) =>
   789→        `${c.name || ""} ${c.nameEn || ""} ${c.rawInput}`
   790→          .toLowerCase()
   791→          .includes(q),
   792→      );
   793→    }

   795→      list = [...list].sort((a, b) =>
   796→        (a.name || a.rawInput).localeCompare(b.name || b.rawInput, "fa"),
   797→      );
   798→    } else if (sortBy === "price-asc") {
   799→      list = [...list].sort((a, b) => (a.priceValue || 0) - (b.priceValue || 0));
   800→    } else if (sortBy === "price-desc") {
   801→      list = [...list].sort((a, b) => (b.priceValue || 0) - (a.priceValue || 0));

   829→
   830→  return (
   831→    <TooltipProvider delayDuration={200}>

   842→            aria-hidden
   843→          />

   845→            className="pointer-events-none absolute -bottom-12 -right-8 size-32 rounded-full bg-amber-200/40 blur-2xl dark:bg-amber-500/10"
   846→            aria-hidden
