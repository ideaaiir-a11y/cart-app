    69→function ThemeMiniPreview({ swatch, dark }: { swatch: string[]; dark?: boolean }) {
    70→  const [accent, soft, usage] = swatch;
    71→  const bg = dark ? soft : "#ffffff";
    72→  return (
    73→    <div
    74→      className="w-full rounded-lg border p-2.5 shadow-sm"
    75→      style={{ background: bg, borderColor: dark ? "#2b3a52" : "#e5e7eb" }}
    76→      aria-hidden
    77→    >
    78→      <div className="mb-1.5 flex items-center gap-1.5">
    79→        <span className="h-1.5 w-10 rounded-full" style={{ background: dark ? "#8fa1ba" : "#cbd5e1" }} />
    80→        <span className="h-1.5 w-6 rounded-full" style={{ background: dark ? "#3b4c66" : "#e2e8f0" }} />
    81→      </div>
    82→      <div className="mb-1.5 h-8 rounded-md" style={{ background: dark ? "#223047" : "#f1f5f9" }} />
    83→      <div className="flex items-center justify-between gap-2">
    84→        <span
    85→          className="rounded-full px-2 py-0.5 text-[9px] font-bold"
    86→          style={{ background: dark ? "rgba(255,255,255,.08)" : soft, color: accent }}
    87→        >
    88→          ۱۲۳٬۰۰۰
    89→        </span>
    90→        <span className="h-1.5 w-12 rounded-full" style={{ background: usage, opacity: 0.55 }} />
    91→      </div>
    92→    </div>
    93→  );
    94→}
    95→
    96→export function SettingsSection({
    97→  watermark,
    98→  onWatermarkChange,
    99→  assistant,
   100→  onAssistantChange,
   101→  cardTheme,
   102→  onCardThemeChange,
   103→  disabled,
   104→}: SettingsSectionProps) {
   105→  const [saving, setSaving] = React.useState(false);
   106→  const [baleToken, setBaleToken] = React.useState("");
   107→  const [showToken, setShowToken] = React.useState(false);
   108→  const [baleLoading, setBaleLoading] = React.useState(false);
   109→  const [baleConnected, setBaleConnected] = React.useState(false);
   110→  const [baleInfo, setBaleInfo] = React.useState<BaleInfo | null>(null);
   111→  const [baleError, setBaleError] = React.useState("");
   112→  const [copied, setCopied] = React.useState(false);
   113→
   114→  /* پیش‌نمایش زنده تم */
   115→  const [previewTheme, setPreviewTheme] = React.useState(cardTheme);
   116→  const [previewHtml, setPreviewHtml] = React.useState("");
   117→  const [previewLoading, setPreviewLoading] = React.useState(false);
   118→  const previewCache = React.useRef<Map<string, string>>(new Map());
   119→
   120→  // همگام‌سازی تم پیش‌نمایش با تم انتخابی
   121→  React.useEffect(() => {
   122→    setPreviewTheme(cardTheme);
   123→  }, [cardTheme]);
   124→
   125→  // بارگذاری HTML پیش‌نمایش برای تم فعال (با کش)
   126→  React.useEffect(() => {
   127→    let cancelled = false;
   128→    const cached = previewCache.current.get(previewTheme);
   129→    if (cached) {
   130→      setPreviewHtml(cached);
   131→      return;
   132→    }
   133→    setPreviewLoading(true);
   134→    apiJson<{ ok: boolean; html: string }>(
   135→      `/api/card-preview?theme=${encodeURIComponent(previewTheme)}`,
   136→    )
   137→      .then((data) => {
   138→        if (cancelled) return;
   139→        previewCache.current.set(previewTheme, data.html || "");
   140→        setPreviewHtml(data.html || "");
   141→      })
   142→      .catch(() => {
   143→        if (!cancelled) setPreviewHtml("");
   144→      })
   145→      .finally(() => {
   146→        if (!cancelled) setPreviewLoading(false);
   147→      });
   148→    return () => {
   149→      cancelled = true;
   150→    };
   151→  }, [previewTheme]);
   152→
   153→  // بررسی وضعیت اتصال قبلی ربات بله
   154→  React.useEffect(() => {
   155→    let cancelled = false;
   156→    apiJson<{ ok: boolean; tokenSet: boolean; lastInfo?: BaleInfo }>(
   157→      "/api/bale/status",
   158→    )
   159→      .then((data) => {
   160→        if (cancelled) return;
   161→        setBaleConnected(Boolean(data.tokenSet));
   162→        if (data.lastInfo) setBaleInfo(data.lastInfo);
   163→      })
   164→      .catch(() => {});
   165→    return () => {
   166→      cancelled = true;
   167→    };
   168→  }, []);
   169→
   170→  const posLabel =
   171→    WATERMARK_POSITIONS.find((p) => p.value === watermark.pos)?.label ?? "";
   172→
   173→  const saveSettings = async () => {
   174→    setSaving(true);
   175→    try {
   176→      await apiJson("/api/settings", {
   177→        method: "POST",
   178→        headers: { "Content-Type": "application/json" },
   179→        body: JSON.stringify({
   180→          watermarkText: watermark.text,
   181→          watermarkMode: watermark.mode,
   182→          watermarkPos: watermark.pos,
   183→          cardTheme,
   184→          assistantUrl: assistant.url,
   185→          assistantKey: assistant.key,
   186→          assistantModel: assistant.model,
   187→          useExternal: assistant.enabled,
   188→        }),
   189→      });
   190→      toast.success("تنظیمات با موفقیت ذخیره شد");
   191→    } catch (e) {
   192→      toast.error("ذخیره تنظیمات ناموفق بود", {
   193→        description: toErrorMessage(e),
   194→      });
   195→    } finally {
   196→      setSaving(false);
   197→    }
   198→  };
   199→
   200→  /* ذخیره فوری تم هنگام انتخاب */
   201→  const pickTheme = async (key: string) => {
   202→    onCardThemeChange(key);
   203→    try {
   204→      await apiJson("/api/settings", {
   205→        method: "POST",
   206→        headers: { "Content-Type": "application/json" },
   207→        body: JSON.stringify({ cardTheme: key }),
   208→      });
   209→    } catch {
   210→      /* بی‌صدا — انتخاب محلی حفظ می‌شود */
   211→    }
   212→  };
   213→
   214→  const embedOrigin =
   215→    typeof window !== "undefined" ? window.location.origin : "https://your-domain.com";
   216→  const embedSnippet = `<script\n  src="${embedOrigin}/api/embed/widget.js"\n  data-title="پشتیبانی فروشگاه"\n  data-pos="left"\n  data-accent="#059669"\n  defer\n></script>`;
   217→
   218→  const copySnippet = async () => {
   219→    try {
   220→      await navigator.clipboard.writeText(embedSnippet);
   221→      setCopied(true);
   222→      toast.success("کد ویجت کپی شد");
   223→      window.setTimeout(() => setCopied(false), 2000);
   224→    } catch (e) {
   225→      toast.error("کپی ناموفق بود", { description: toErrorMessage(e) });
   226→    }
   227→  };
   228→
   229→  const setupBale = async () => {
   230→    const token = baleToken.trim();
   231→    if (!token) {
   232→      toast.error("توکن ربات بله را وارد کنید");
   233→      return;
   234→    }
   235→    setBaleLoading(true);
   236→    setBaleError("");
   237→    try {
   238→      const data = await apiJson<{
   239→        ok: boolean;
   240→        info?: BaleInfo;
   241→        error?: string;
   242→      }>("/api/bale/setup", {
   243→        method: "POST",
   244→        headers: { "Content-Type": "application/json" },
   245→        body: JSON.stringify({ token }),
   246→      });
   247→      if (data.ok) {
   248→        setBaleConnected(true);
   249→        setBaleInfo(data.info ?? null);
   250→        setBaleToken("");
   251→        toast.success("ربات بله متصل شد و وب‌هوک ثبت گردید");
   252→      } else {
   253→        setBaleConnected(false);
   254→        setBaleError(data.error ?? "اتصال ناموفق بود");
   255→        toast.error("اتصال ربات بله ناموفق بود", {
   256→          description: data.error,
   257→        });
   258→      }
   259→    } catch (e) {
   260→      setBaleConnected(false);
   261→      setBaleError(toErrorMessage(e));
   262→      toast.error("اتصال ربات بله ناموفق بود", {
   263→        description: toErrorMessage(e),
   264→      });
   265→    } finally {
   266→      setBaleLoading(false);
   267→    }
   268→  };
   269→
   270→  const readLogo = (file: File) => {
   271→    const reader = new FileReader();
   272→    reader.onload = () => {
   273→      onWatermarkChange({
   274→        ...watermark,
   275→        logoDataUrl: String(reader.result ?? ""),
   276→        logoName: file.name,
   277→      });
   278→    };
   279→    reader.readAsDataURL(file);
   280→  };
   281→
   282→  return (
   283→    <Card
   284→      id="settings"
   285→      className={`scroll-mt-24 ${disabled ? "pointer-events-none opacity-60" : undefined}`}
   286→    >
   287→      <CardHeader>
   288→        <CardTitle className="flex items-center gap-2 text-lg">
   289→          <span className="flex size-9 items-center justify-center rounded-xl bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400">
   290→            <Stamp className="size-5" aria-hidden />
   291→          </span>
   292→          تنظیمات
   293→        </CardTitle>
   294→        <CardDescription>
   295→          واترمارک، تم قالب، دستیار خارجی و ربات بله را پیکربندی کنید
   296→        </CardDescription>
   297→      </CardHeader>
   298→
   299→      <CardContent>
   300→        <Tabs defaultValue="watermark">
   301→          <TabsList className="grid h-11 w-full grid-cols-4 rounded-xl">
   302→            <TabsTrigger value="watermark" className="gap-1 rounded-lg px-1.5">
   303→              <Stamp className="size-4" aria-hidden />
   304→              <span className="text-xs">واترمارک</span>
   305→            </TabsTrigger>
   306→            <TabsTrigger value="theme" className="gap-1 rounded-lg px-1.5">
   307→              <Palette className="size-4" aria-hidden />
   308→              <span className="text-xs">قالب کارت</span>
   309→            </TabsTrigger>
   310→            <TabsTrigger value="assistant" className="gap-1 rounded-lg px-1.5">
   311→              <Bot className="size-4" aria-hidden />
   312→              <span className="text-xs">دستیار</span>
   313→            </TabsTrigger>
   314→            <TabsTrigger value="bale" className="gap-1 rounded-lg px-1.5">
   315→              <Send className="size-4" aria-hidden />
   316→              <span className="text-xs">ربات بله</span>
   317→            </TabsTrigger>
   318→          </TabsList>
   319→
   320→          {/* ── تب واترمارک ── */}
   321→          <TabsContent value="watermark" className="mt-4 space-y-5">
   322→            <div className="flex items-center justify-between gap-3 rounded-xl border bg-muted/40 p-3.5">
   323→              <div className="space-y-0.5">
   324→                <Label htmlFor="logo-mode-switch" className="text-sm font-semibold">
   325→                  استفاده از لوگو به‌جای متن
   326→                </Label>
   327→                <p className="text-xs text-muted-foreground">
   328→                  {watermark.mode === "text"
   329→                    ? "واترمارک متنی روی تصویر اعمال می‌شود"
   330→                    : "تصویر لوگوی شما روی تصویر اعمال می‌شود"}
   331→                </p>
   332→              </div>
   333→              <Switch
   334→                id="logo-mode-switch"
   335→                checked={watermark.mode === "logo"}
   336→                onCheckedChange={(v) =>
   337→                  onWatermarkChange({ ...watermark, mode: v ? "logo" : "text" })
   338→                }
   339→                aria-label="تغییر حالت واترمارک بین متن و لوگو"
   340→              />
   341→            </div>
   342→
   343→            {watermark.mode === "text" ? (
   344→              <div className="space-y-4">
   345→                <div className="space-y-2">
   346→                  <Label htmlFor="watermark-text" className="text-sm font-semibold">
   347→                    متن واترمارک
   348→                  </Label>
   349→                  <Input
   350→                    id="watermark-text"
   351→                    value={watermark.text}
   352→                    onChange={(e) =>
   353→                      onWatermarkChange({ ...watermark, text: e.target.value })
   354→                    }
   355→                    placeholder="مثلاً: فروشگاه آرایشی من"
   356→                    className="h-11 rounded-xl"
   357→                  />
   358→                </div>
   359→                <div className="space-y-2">
   360→                  <Label className="text-sm font-semibold">موقعیت واترمارک</Label>
   361→                  <Select
   362→                    dir="rtl"
   363→                    value={watermark.pos}
   364→                    onValueChange={(v) => onWatermarkChange({ ...watermark, pos: v })}
   365→                  >
   366→                    <SelectTrigger className="h-11 w-full rounded-xl">
   367→                      <SelectValue placeholder="انتخاب موقعیت" />
   368→                    </SelectTrigger>
   369→                    <SelectContent>
   370→                      {WATERMARK_POSITIONS.map((p) => (
   371→                        <SelectItem key={p.value} value={p.value}>
   372→                          {p.label}
   373→                        </SelectItem>
   374→                      ))}
   375→                    </SelectContent>
   376→                  </Select>
   377→                </div>
   378→              </div>
   379→            ) : (
   380→              <div className="space-y-3">
   381→                <Label className="text-sm font-semibold">تصویر لوگو</Label>
   382→                {watermark.logoDataUrl ? (
   383→                  <div className="flex items-center gap-3 rounded-xl border bg-muted/40 p-3">
   384→                    <img
   385→                      src={watermark.logoDataUrl}
   386→                      alt="پیش‌نمایش لوگو"
   387→                      className="size-14 rounded-lg border bg-white object-contain p-1"
   388→                    />
   389→                    <div className="min-w-0 flex-1">
   390→                      <p className="truncate text-sm font-medium" dir="ltr">
   391→                        {watermark.logoName}
   392→                      </p>
   393→                      <p className="text-xs text-muted-foreground">
   394→                        لوگو بارگذاری شد
   395→                      </p>
   396→                    </div>
   397→                    <Button
   398→                      type="button"
   399→                      variant="ghost"
   400→                      size="icon"
   401→                      className="size-9 shrink-0 rounded-lg text-red-500 hover:bg-red-50 hover:text-red-600 dark:hover:bg-red-500/10"
   402→                      onClick={() =>
   403→                        onWatermarkChange({
   404→                          ...watermark,
   405→                          logoDataUrl: "",
   406→                          logoName: "",
   407→                        })
   408→                      }
   409→                      aria-label="حذف لوگو"
   410→                    >
   411→                      <Trash2 className="size-4" aria-hidden />
   412→                    </Button>
   413→                  </div>
   414→                ) : (
   415→                  <label className="flex min-h-24 cursor-pointer flex-col items-center justify-center gap-1.5 rounded-xl border-2 border-dashed border-border bg-muted/40 p-4 text-center transition-colors hover:border-emerald-400 hover:bg-emerald-50/60 dark:hover:bg-emerald-950/20">
   416→                    <ImagePlus className="size-6 text-muted-foreground" aria-hidden />
   417→                    <span className="text-sm text-muted-foreground">
   418→                      انتخاب تصویر لوگو (PNG یا SVG شفاف)
   419→                    </span>
   420→                    <input
   421→                      type="file"
   422→                      accept="image/png,image/jpeg,image/webp,image/svg+xml"
   423→                      className="hidden"
   424→                      onChange={(e) => {
   425→                        const file = e.target.files?.[0];
   426→                        if (file) readLogo(file);
   427→                        e.target.value = "";
   428→                      }}
   429→                    />
   430→                  </label>
   431→                )}
   432→              </div>
   433→            )}
   434→
   435→            <p className="rounded-xl bg-amber-50 px-3.5 py-2.5 text-xs leading-6 text-amber-800 dark:bg-amber-950/30 dark:text-amber-300">
   436→              راهنما: {watermark.mode === "text" ? `متن «${watermark.text || "برند شما"}» ` : "لوگوی انتخابی "}
   437→              به‌صورت <b>{posLabel}</b> روی تصویر اصلی هر کارت اعمال می‌شود.
   438→            </p>
   439→          </TabsContent>
   440→
   441→          {/* ── تب قالب کارت (تم‌های رنگی + پیش‌نمایش زنده) ── */}
   442→          <TabsContent value="theme" className="mt-4 space-y-4">
   443→            <p className="text-xs leading-6 text-muted-foreground">
   444→              تم رنگی قالب کارت‌های خروجی را انتخاب کنید؛ رنگ قیمت، جدول مشخصات،
   445→              روش مصرف و دکمه لینک مطابق تم رندر می‌شود. انتخاب فوراً ذخیره می‌شود.
   446→            </p>
   447→            <div className="grid grid-cols-2 gap-2.5">
   448→              {CARD_THEME_OPTIONS.map((t) => {
   449→                const active = cardTheme === t.key;
   450→                const hovered = previewTheme === t.key;
   451→                return (
   452→                  <button
   453→                    key={t.key}
   454→                    type="button"
   455→                    onClick={() => void pickTheme(t.key)}
   456→                    onMouseEnter={() => setPreviewTheme(t.key)}
   457→                    onFocus={() => setPreviewTheme(t.key)}
   458→                    aria-pressed={active}
   459→                    className={`group relative flex flex-col gap-2 rounded-xl border-2 p-2.5 text-right outline-none transition-all hover:-translate-y-0.5 hover:shadow-md focus-visible:ring-2 focus-visible:ring-emerald-500 ${
   460→                      active
   461→                        ? "border-emerald-500 bg-emerald-50/70 shadow-sm dark:bg-emerald-950/30"
   462→                        : hovered
   463→                          ? "border-emerald-300 bg-card shadow-sm"
   464→                          : "border-border bg-card hover:border-emerald-300"
   465→                    }`}
   466→                  >
   467→                    {active && (
   468→                      <CheckCircle2
   469→                        className="absolute left-2 top-2 size-4 text-emerald-600 dark:text-emerald-400"
   470→                        aria-hidden
   471→                      />
   472→                    )}
   473→                    <ThemeMiniPreview swatch={t.swatch} dark={t.key === "midnight"} />
   474→                    <span className="flex items-center gap-1.5 text-xs font-bold">
   475→                      <span
   476→                        className="inline-block size-3 rounded-full ring-1 ring-black/10"
   477→                        style={{ background: t.swatch[0] }}
   478→                        aria-hidden
   479→                      />
   480→                      {t.label}
   481→                    </span>
   482→                  </button>
   483→                );
   484→              })}
   485→            </div>
   486→
   487→            {/* پیش‌نمایش زنده قالب */}
   488→            <div className="overflow-hidden rounded-xl border shadow-sm">
   489→              <div className="flex items-center justify-between gap-2 border-b bg-muted/60 px-3 py-2">
   490→                <div className="flex items-center gap-1.5" aria-hidden>
   491→                  <span className="size-2.5 rounded-full bg-red-400" />
   492→                  <span className="size-2.5 rounded-full bg-amber-400" />
   493→                  <span className="size-2.5 rounded-full bg-emerald-400" />
   494→                </div>
   495→                <p className="flex items-center gap-1.5 text-[11px] font-bold text-muted-foreground">
   496→                  <MonitorPlay className="size-3.5" aria-hidden />
   497→                  پیش‌نمایش زنده — تم {CARD_THEME_OPTIONS.find((t) => t.key === previewTheme)?.label ?? ""}
   498→                </p>
   499→              </div>
   500→              <div
   501→                className="relative flex h-[300px] justify-center overflow-hidden"
   502→                style={{
   503→                  background:
   504→                    previewTheme === "midnight" ? "#0b1220" : "#f3f4f6",
   505→                }}
   506→              >
   507→                {previewHtml ? (
   508→                  <iframe
   509→                    srcDoc={previewHtml}
   510→                    title="پیش‌نمایش زنده قالب کارت"
   511→                    sandbox=""
   512→                    loading="lazy"
   513→                    className="pointer-events-none h-[764px] w-[480px] shrink-0 border-0 bg-white"
   514→                    style={{ transform: "scale(0.62)", transformOrigin: "top center" }}
   515→                  />
   516→                ) : (
