    78→    78→function buildCardsCsv(cards: CardWithHtml[]): string {
    79→    79→  const header = [
    80→    80→    "نام محصول",
    81→    81→    "نام لاتین",
    82→    82→    "قیمت (تومان)",
    83→    83→    "لینک مرجع",
    84→    84→    "تم قالب",
    85→    85→    "تعداد مشخصات",
    86→    86→    "روش مصرف",
    87→    87→  ];
    88→    88→  const rows = cards.map((c) => [
    89→    89→    c.name || c.rawInput,
    90→    90→    c.nameEn || "",
    91→    91→    c.priceValue || 0,
    92→    92→    c.link || "",
    93→    93→    themeLabel(c.theme),
    94→    94→    c.specs?.length ?? 0,
    95→    95→    (c.usage || "").replace(/\r?\n/g, " | "),
    96→    96→  ]);
    97→    97→  const body = [header, ...rows].map((r) => r.map(csvCell).join(",")).join("\r\n");
    98→    98→  return `\uFEFF${body}`;
    99→    99→}

   428→   428→
   429→   429→function BulkThemePopover({
   430→   430→  jobId,
   431→   431→  disabled,
   432→   432→  onApplied,
   433→   433→}: {
   434→   434→  jobId: string;
   435→   435→  disabled: boolean;
   436→   436→  onApplied: () => void;
   437→   437→}) {
   438→   438→  const [open, setOpen] = React.useState(false);
   439→   439→  const [applying, setApplying] = React.useState("");
   440→   440→
   441→   441→  const applyTheme = async (key: string, label: string) => {
   442→   442→    setApplying(key);
   443→   443→    try {
   444→   444→      const data = await apiJson<{ ok: boolean; updated: number; theme: string }>(
   445→   445→        `/api/jobs/${jobId}/theme`,
   446→   446→        {
   447→   447→          method: "POST",
   448→   448→          headers: { "Content-Type": "application/json" },
   449→   449→          body: JSON.stringify({ theme: key }),
   450→   450→        },
   451→   451→      );
   452→   452→      toast.success(`تم «${label}» روی ${faNum(data.updated ?? 0)} کارت اعمال شد`);
   453→   453→      setOpen(false);

   455→   455→    } catch (e) {
